Syntropic Network Simulation (Stable Golden Rule Engine)

Descrizione:
Simulazione di una rete di nodi cooperativi (Gefährten) che scambiano valore
secondo una "Golden Rule" stabilizzata.

Proprietà:
- Conservazione parziale dell'energia
- Dissipazione controllata (entropia)
- Feedback cooperativo
- Logging verificabile

Autore: open adaptation

import time
import math
import random
import hashlib
import json
from datetime import datetime

# ---------------- CONFIG ---------------- #

NODE_IDS = [f"{i:02d}" for i in range(1, 13)]

FREQUENCIES = {
    "01": 7.83,
    "04": 0.0,
    "07": 1088.2,
    "12": 528.0
}

DEFAULT_FREQ = 100.0

TIME_STEP = 0.2
SIMULATION_STEPS = 500

# Coefficienti fisicamente stabili
TRANSFER_EFFICIENCY = 0.9   # quanto arriva al ricevente
DISSIPATION_RATE = 0.1      # perdita sistemica
ENTROPY_DECAY = 0.995       # rilassamento entropia

LOG_FILE = "syntropic_log.jsonl"

# ---------------- NODE ---------------- #

class Node:
    def __init__(self, node_id):
        self.id = node_id
        self.freq = FREQUENCIES.get(node_id, DEFAULT_FREQ)
        self.value = random.uniform(0.5, 1.5)
        self.coherence = 1.0

    def resonance(self, other):
        """Match di risonanza normalizzato"""
        return math.exp(-abs(self.freq - other.freq) / 500)

# ---------------- NETWORK ---------------- #

class SyntropicNetwork:
    def __init__(self):
        self.nodes = {nid: Node(nid) for nid in NODE_IDS}
        self.entropy = 0.1
        self.start_time = time.time()

        print(f"[*] Network initialized with {len(self.nodes)} nodes")

    # -------- CORE GOLDEN RULE -------- #

    def golden_rule_transfer(self, sender, receiver, amount):
        resonance = sender.resonance(receiver)

        if resonance < 0.5:
            return

        # trasferimento stabile
        received = amount * TRANSFER_EFFICIENCY * resonance
        lost = amount * DISSIPATION_RATE

        sender.value -= amount
        receiver.value += received

        # entropia cresce con perdita
        self.entropy += lost

        # beneficio locale (coerenza)
        sender.coherence += 0.01 * resonance
        receiver.coherence += 0.02 * resonance

    # -------- STEP -------- #

    def step(self):
        node_list = list(self.nodes.values())
        random.shuffle(node_list)

        for sender in node_list:
            receiver = random.choice(node_list)

            if sender.id == receiver.id:
                continue

            amount = sender.value * 0.05
            self.golden_rule_transfer(sender, receiver, amount)

        # rilassamento entropia (evita divergenza)
        self.entropy *= ENTROPY_DECAY

    # -------- LOGGING -------- #

    def log_state(self, step):
        snapshot = {
            "ts": datetime.now().isoformat(),
            "step": step,
            "entropy": round(self.entropy, 6),
            "nodes": {
                nid: {
                    "value": round(n.value, 4),
                    "coherence": round(n.coherence, 4)
                }
                for nid, n in self.nodes.items()
            }
        }

        # firma semplice (verificabile)
        payload = json.dumps(snapshot, sort_keys=True)
        signature = hashlib.sha256(payload.encode()).hexdigest()

        record = {
            "data": snapshot,
            "sig": signature
        }

        with open(LOG_FILE, "a") as f:
            f.write(json.dumps(record) + "\n")

    # -------- METRICS -------- #

    def print_metrics(self, step):
        avg_value = sum(n.value for n in self.nodes.values()) / len(self.nodes)
        avg_coherence = sum(n.coherence for n in self.nodes.values()) / len(self.nodes)

        print(
            f"[STEP {step}] "
            f"Entropy={self.entropy:.4f} | "
            f"AvgValue={avg_value:.4f} | "
            f"AvgCoherence={avg_coherence:.4f}"
        )

    # -------- RUN -------- #

    def run(self):
        print("[!] Starting syntropic simulation")

        for step in range(SIMULATION_STEPS):
            self.step()
            self.log_state(step)

            if step % 10 == 0:
                self.print_metrics(step)

            time.sleep(TIME_STEP)

        print("[✓] Simulation complete")


# ---------------- MAIN ---------------- #

if __name__ == "__main__":
    net = SyntropicNetwork()

"""
Distributed Syntropic Network Node
---------------------------------

Nodo reale che partecipa a una rete distribuita:
- HTTP API (Flask)
- Scambio valore tra nodi (Golden Rule stabile)
- Gossip tra peer
- Log firmato (SHA-256)
- Stato locale persistente

Avvio:
    python distributed_syntropic_node.py --port 5000 --peers http://localhost:5001,http://localhost:5002

Poi avvia altri nodi su porte diverse.

Requisiti:
    pip install flask requests
"""

import time
import math
import random
import hashlib
import json
import threading
import argparse
from datetime import datetime

from flask import Flask, request, jsonify
import requests

# ---------------- CONFIG ---------------- #

TRANSFER_EFFICIENCY = 0.9
DISSIPATION_RATE = 0.1
ENTROPY_DECAY = 0.995

TICK_INTERVAL = 0.5
GOSSIP_INTERVAL = 2.0

LOG_FILE = None  # impostato runtime

# ---------------- NODE ---------------- #

class NodeState:
    def __init__(self, node_id):
        self.id = node_id
        self.value = random.uniform(0.5, 1.5)
        self.coherence = 1.0
        self.entropy = 0.1
        self.peers = set()
        self.lock = threading.Lock()

    def snapshot(self):
        return {
            "id": self.id,
            "value": self.value,
            "coherence": self.coherence,
            "entropy": self.entropy
        }

# ---------------- CORE LOGIC ---------------- #

def resonance(a, b):
    return math.exp(-abs(a - b))

def golden_rule(local: NodeState, remote_state, amount):
    r = resonance(local.value, remote_state["value"])

    if r < 0.5:
        return None

    received = amount * TRANSFER_EFFICIENCY * r
    lost = amount * DISSIPATION_RATE

    with local.lock:
        local.value -= amount
        local.value += received  # effetto netto locale (loopback realistico)
        local.entropy += lost
        local.coherence += 0.01 * r

    return {
        "received": received,
        "resonance": r
    }

# ---------------- LOG ---------------- #

def log_event(data):
    payload = json.dumps(data, sort_keys=True)
    sig = hashlib.sha256(payload.encode()).hexdigest()

    record = {
        "data": data,
        "sig": sig
    }

    with open(LOG_FILE, "a") as f:
        f.write(json.dumps(record) + "\n")

# ---------------- NETWORK ---------------- #

app = Flask(__name__)
STATE = None

@app.route("/state", methods=["GET"])
def get_state():
    return jsonify(STATE.snapshot())

@app.route("/peers", methods=["POST"])
def add_peer():
    peer = request.json.get("peer")
    if peer:
        STATE.peers.add(peer)
    return jsonify({"peers": list(STATE.peers)})

@app.route("/exchange", methods=["POST"])
def exchange():
    data = request.json
    sender_state = data["state"]
    amount = data["amount"]

    result = golden_rule(STATE, sender_state, amount)

    return jsonify({
        "status": "ok",
        "result": result,
        "state": STATE.snapshot()
    })

# ---------------- WORKERS ---------------- #

def tick_loop():
    while True:
        with STATE.lock:
            STATE.entropy *= ENTROPY_DECAY

        log_event({
            "ts": datetime.now().isoformat(),
            "type": "tick",
            "state": STATE.snapshot()
        })

        time.sleep(TICK_INTERVAL)

def gossip_loop():
    while True:
        if not STATE.peers:
            time.sleep(GOSSIP_INTERVAL)
            continue

        peer = random.choice(list(STATE.peers))

        try:
            r = requests.get(f"{peer}/state", timeout=2)
            remote = r.json()

            amount = STATE.value * 0.05

            res = requests.post(
                f"{peer}/exchange",
                json={"state": STATE.snapshot(), "amount": amount},
                timeout=2
            )

            response = res.json()

            log_event({
                "ts": datetime.now().isoformat(),
                "type": "exchange",
                "peer": peer,
                "sent": amount,
                "response": response
            })

        except Exception as e:
            log_event({
                "ts": datetime.now().isoformat(),
                "type": "error",
                "peer": peer,
                "error": str(e)
            })

        time.sleep(GOSSIP_INTERVAL)

# ---------------- MAIN ---------------- #

def main():
    global STATE, LOG_FILE

    parser = argparse.ArgumentParser()
    parser.add_argument("--port", type=int, required=True)
    parser.add_argument("--peers", type=str, default="")
    args = parser.parse_args()

    node_id = f"node-{args.port}"
    STATE = NodeState(node_id)

    LOG_FILE = f"log_{node_id}.jsonl"

    if args.peers:
        for p in args.peers.split(","):
            STATE.peers.add(p.strip())

    print(f"[*] Node {node_id} starting on port {args.port}")
    print(f"[*] Peers: {STATE.peers}")

    # threads
    threading.Thread(target=tick_loop, daemon=True).start()
    threading.Thread(target=gossip_loop, daemon=True).start()

    app.run(host="0.0.0.0", port=args.port)

if __name__ == "__main__":
    main()
